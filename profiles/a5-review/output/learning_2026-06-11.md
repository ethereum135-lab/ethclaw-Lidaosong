```python
def sun_narrative_freshness_check(symbol, narrative_type, first_pump_date,
                                   volume_history, repeat_buyers_pct):
    """
    孙宇晨式叙事新鲜度检查 — 这个故事的'保鲜期'判断
    
    叙事类型 & 标准寿命:
    - 🟢 协议升级/主网上线: 14-21天（最长的保质期）
    - 🟢 机构合作/合规利好: 7-14天
    - 🟡 板块轮动: 5-7天
    - 🟡 名人喊单/Twitter热度: 2-3天
    - 🔴 Meme/纯热点: 1-2天（最长3天）
    """
    narrative_lifespan = {
        "生态升级": 18,     # 最可靠
        "协议升级": 18,
        "主网上线": 14,
        "机构合作": 10,
        "合规利好": 10,
        "板块轮动": 6,
        "名人喊单": 3,
        "Twitter热度": 2,
        "Meme": 2,
        "AI叙事": 5,
        "空投预期": 7,
        "回购销毁": 10,
    }
    
    typical_days = narrative_lifespan.get(narrative_type, 5)
    days_since_first_pump = days_since(first_pump_date)
    remaining = typical_days - days_since_first_pump
    
    if remaining <= 0:
        return f"⛔ 叙事已过期 — '{narrative_type}'标准周期{typical_days}天，本轮已{days_since_first_pump}天。孙宇晨会说：'这个故事该翻篇了，市场已经price in。你看到的价格不是价值发现，是最后的流动性在退出。'"
    elif remaining <= 2:
        return f"🔴 即将过期 — 还剩{remaining}天。孙宇晨会准备退出：不要把价值投资和过期叙事搞混。"
    elif remaining <= 5:
        return f"🟡 稳定期 — 还剩{remaining}天。可以持有但准备止损。孙宇晨不会在这个阶段加仓。"
    else:
        return f"🟢 新鲜期 — 还剩{remaining}天。孙宇晨会在这个阶段重点关注并择机入场。"
```

### 发现2：F&G≤15极恐环境中，A4买入信号应叠加「孙宇晨第二天确认过滤」

**问题：** 在F&G=9的极恐环境中，单日逆势上涨的币中超过60%是一日游反弹（数据参考：A9第24周报告中BTC Dorminance 60%+资金不扩散）。A4当前在第一天的买入信号发出后即入场，没有等待24h确认。

**孙宇晨规则建议：** 「极恐环境中，趋势需要**双重确认**——第一天放量上涨→标记观察。第二天低开高走或低开快速拉回→确认买盘真实→入场。这是孙宇晨和散户的根本区别——散户怕错过，孙宇晨不怕。」

```python
def sun_fear_confirmation_filter(symbol, day1_gain_pct, day1_volume_ratio,
                                   day2_open_vs_close, fg_index):
    """孙宇晨式极恐双重确认过滤"""
    if fg_index >= 40:
        return "PASS — 正常环境，常规入场逻辑"
    
    if fg_index < 15:
        # 极恐环境：需要双重确认
        if day2_open_vs_close == 'LOWER_THEN_RECOVER':
            return f"✅ 双重确认通过 — {symbol}第一天+{day1_gain_pct}%，第二天低开但拉回→确认买盘真实。入场。"
        elif day2_open_vs_close == 'CONTINUE_UP':
            return f"🟡 强势延续 — {symbol}连续两天上涨。可入场但注意：连续大涨后回调风险增加。限仓50%。"
        elif day2_open_vs_close == 'LOWER_ALL_DAY':
            return f"⛔ 双重确认失败 — {symbol}第二天低开低走，第一天上涨被证实为一日游反弹。等待下一次信号。"
        else:
            return f"⏳ 等待第二天确认"
```

**影响：** 如果这个规则应用于今天——C在06-11第一天+9.76%，系统应标记为「观察，等06-12第二天确认」，而不是今天直接入场。如果06-12 C回踩确认→入场时机更安全。如果06-12 C直接跌回去→避开了亏损。
